# Требования к зависимостям у 3.31 те же, что у 3.30 (aiohttp >=3.9,<3.15;

aiogram==3.31.0
# aiohttp: 3.9.5 → 3.14.3 ради безопасности (pip-audit, сентябрь 2026: 34
# уязвимости в 3.9.5, из них HIGH — CVE-2026-69244, чтение за границей буфера в
# C-парсере ОТВЕТА на кривом chunked, и CVE-2025-69223, zip-бомба). У нас
# aiohttp — только КЛИЕНТ (Bot API через aiogram и notifier, cbu.uz, /healthz),
# серверная часть не используется, но клиентских дыр в списке хватает: утечка
# Cookie при кросс-доменном редиректе, неограниченный DNS-кэш коннектора,
# обход проверки SNI при переиспользовании соединения. 3.14.3 — последняя в
# пределах aiogram 3.31 (<3.15) и первая, где закрыто всё известное.
# Поднимаете — сверяйте с `pip-audit -r requirements.txt` и с aioresponses
# в requirements-dev.txt: он собирает ClientResponse сам, и на 3.14 его пришлось
# подпереть шимом в tests/conftest.py.
aiohttp==3.14.3
# fastapi 0.115.0 → 0.141.1 — не ради самого fastapi, а ради starlette под ним:
# в 0.38.6 семь уязвимостей, последние закрыты только в 1.3.1. Для нас
# существенна CVE-2026-48710 — подменённый Host отравляет request.url.path, а
# на нём метрики /api/* и путь в алерте об ошибке. Остальные (DoS на разборе
# форм, StaticFiles на Windows, HTTPEndpoint) у нас не достижимы: форм нет и
# python-multipart не стоит, сервер на Linux, HTTPEndpoint не используется —
# но и держаться за дырявую версию незачем. fastapi до 0.133 требует
# starlette <1.0, поэтому двигается и он. Из API, убранного в Starlette 1.0, у
# нас был только `@app.on_event("shutdown")` — переведён на lifespan.
# starlette пиним ЯВНО: fastapi 0.141 требует лишь >=0.46 без потолка, и без
